    <button title="el murciélago">Speak a word</button>

    <script>
        const synth = speechSynthesis;
        let voices = [];

        function loadVoices() {
            voices = synth.getVoices();
            console.log('Голоса загружены:', voices.map(v => `${v.name} [${v.lang}]`));
        }

        loadVoices();
        synth.onvoiceschanged = loadVoices;

        function letsSpeak(e) {
            const word = e.target.title;
            console.log('Клик по кнопке, слово:', word, '| голосов:', voices.length);

            const utter = new SpeechSynthesisUtterance(word);

            let chosen = voices.find(v => v.lang.startsWith('es'));

            if (!chosen && voices.length > 0) {
                chosen = voices[0];
                console.warn('Испанский голос не найден, использую:', chosen.name);
            }

            if (chosen) {
                utter.voice = chosen;
                utter.lang = chosen.lang;
            } else {
                utter.lang = 'es';
                console.warn('Голосов вообще нет, ставлю lang=es');
            }

            synth.speak(utter);
        }

        document.querySelectorAll('button')
            .forEach(b => b.addEventListener('click', letsSpeak));
    </script>
