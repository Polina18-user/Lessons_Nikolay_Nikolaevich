# Лабораторная работа №1 
## Задача 1
Написать функцию, которая возвращает кортеж из двух индексов элементов списка lst, таких что сумма элементов по этим индексам равна переменной target. 
Элемент по индексу может быть выбран лишь единожды, значения в списке могут повторяться. Если в списке встречается больше чем два индекса, подходящих под условие - вернуть наименьшие из всех. 
Элементы находятся в списке в произвольном порядке. 

### Программа
main.py
``` python
def two_sum(lst,target):
    if len(lst) > 1:
        for i in range (len(lst)-1):
            for j in range (i+1,len(lst)):
                if lst[i]+lst[j] == target:
                    return (i,j)
    return tuple()

lst=[1,2,3,4,5,6,7,8,9]
target = 8
print(two_sum(lst,target)) # (0,6)
```
test_two_sum.py
``` python
from main import two_sum
import unittest

class TestMath(unittest.TestCase):
    def test_simple(self):
        self.assertEqual(two_sum([1,2,3,4,5,6,7,8,9],8),(0,5))

if __name__ == '__main__':
    unittest.main() # для VS Code
 
```

## Задача 2
### Программа
main.py
``` python
def two_sums_hashed(lst,target):
    dict_seen = {} #словарь для запоминания: число -> индекс
    best_pair = None #лучшая пара
    for i in range (len(lst)):
        num=lst[i] #число их списка
        complement = target - num #находим идеального кандидата, которого бы сюда нужно
        if complement in dict_seen:# если пара уже встречалась
            j = dict_seen[complement] #индекс первого числа
            if best_pair is None or j < best_pair[0]:
                best_pair = (j,i) #запоминаем лучшую пару в правильном порядке наиименьшую
        dict_seen[num] = i #запоминаем текущее число
    return best_pair


lst = [3,3,4,5]
target = 6
print (two_sums_hashed(lst,target))

```

## Задача 3
### Программа
main.py
``` python
def two_sums_hashed(lst,target):
    dict_seen = {} #словарь для запоминания: число -> индекс
    result=[]#пустой список
    for i in range (len(lst)):
        num=lst[i] #число из списка
        complement = target - num #находим идеального кандидата, которого бы сюда нужно
        if complement in dict_seen:# если пара уже встречалась
            for j in dict_seen[complement]:
                result.append((j,i))
        if num in dict_seen:
            dict_seen[num].append(i)
        else:
            dict_seen[num] = [i] 
    return result

lst = [1,2,3,4,5,6,7,8,9]
target = 8
print (two_sums_hashed(lst,target))

```
test_two_sum.py
``` python
from main import two_sums_hashed
import unittest

class TestMath(unittest.TestCase):
    def test_simple(self):
        self.assertEqual(two_sums_hashed([2,2,2],4),[(0,1),(0,2),(1,2)])

if __name__ == '__main__':
    unittest.main() # для VS Code
 
```
