# Лабораторная работа №1 
## Задача 1
Написать функцию, которая возвращает кортеж из двух индексов элементов списка lst, таких что сумма элементов по этим индексам равна переменной target. 
Элемент по индексу может быть выбран лишь единожды, значения в списке могут повторяться. Если в списке встречается больше чем два индекса, подходящих под условие - вернуть наименьшие из всех. 
Элементы находятся в списке в произвольном порядке. 

### Программа
main.py
``` python
def two_sum (lst: list[int], target: int) -> tuple[int,int]| None: 
    """
    Находит индексы двух элементов списка, сумма которых равна target.

    :param lst: список значений
    :type lst: list[int]

    :param target: необходимая сумма в паре
    :type target: int
    :return: кортеж из двух индексов (i,j) либо None
    :rtype: tuple[int, int] | None
    """
    if len(lst) > 1:
        for i in range (len(lst)-1):
            for j in range (i+1,len(lst)):
                if lst[i]+lst[j] == target:
                    return (i,j)
    return None
if __name__ == "__main__":
    lst: list[int] = [1,2,3,4,5,6,7,8,9]
    target: int = 8
    print(two_sum(lst,target)) # (0,6)

```

``` python
def two_sums_hashed(lst: list[int],target: int) -> tuple (int,int)|None:
    """Находит индексы двух элементов списка, сумма которых равна target. 
    Вычислительная сложность алгоритма должна быть ниже чем On^2.
    
    Использует хэш-таблицу (словарь) для поиска комплемента за O(1).
    Итоговая вычислительная сложность - O(n)

    Если подходящих пар несколько, возвращается пара с наименьшим первым
    индексом ( то есть первую по порядку слева)

    :param lst: список целых чисел для поиска
    :type lst: list[int]
    :param target: искомая сумма пары
    :type target: int
    :return: кортеж из двух значений индексов (i,j), где i<j, 
    либо None, если подходящей пары нет
    :rtype: tuple[int,int] | None
    """
    dict_seen:dict[int: int] = {} #словарь для запоминания: число -> индекс
    best_pair: tuple [int, int] | None = None #лучшая пара
    complement: int
    num: int
    for i in range (len(lst)):
        num=lst[i] #число их списка
        complement = target - num #находим идеального кандидата, которого бы сюда нужно
        if complement in dict_seen:# если пара уже встречалась
            j = dict_seen[complement] #индекс первого числа
            if best_pair is None or j < best_pair[0]:
                best_pair = (j,i) #запоминаем лучшую пару в правильном порядке наиименьшую
        dict_seen[num] = i #запоминаем текущее число
    return best_pair

if __name__ == "__main__":
    lst: list[int] = [3,3,4,5]
    target: int = 6
    print (two_sums_hashed(lst,target))#(0,1)

```

### Программа
main.py
``` python
def two_sums_hashed(lst: list[int],target: int) -> list[tuple [int,int]]:
    """Находит индексы двух элементов списка, сумма которых равна target. 
    Вычислительная сложность алгоритма неважна.
    

    Если подходящих пар несколько, возвращаются все пары, если таковых нет пустой список

    :param lst: список целых чисел для поиска
    :type lst: list[int]
    :param target: искомая сумма пары
    :type target: int
    :return: список кортеж из двух  значений индексов (i,j), где i<j, 
    либо пустой, если подходящих пар нет
    :rtype: list(tuple[int,int]) | list(None)
    """
    dict_seen:dict[int: int] = {} #словарь для запоминания: число -> индекс
    result: list[tuple[int,int]]=[]#пустой список


    for i in range (len(lst)):
        num: int=lst[i] #число их списка
        complement: int = target - num #находим идеального кандидата, которого бы сюда нужно
        if complement in dict_seen:# если пара уже встречалась
            for j in dict_seen[complement]:
                result.append((j,i))
        if num in dict_seen:
            dict_seen[num].append(i)
        else:
            dict_seen[num] = [i] 
    return result


if __name__ == "__main__":
    lst: list[int] = [3,3,4,5]
    target: int = 6
    print (two_sums_hashed(lst,target))#[(0,1)]

```

``` python
"""
Тесты для функции two_sums_hashed из модуля main.
"""
import unittest

from main import two_sums_hashed


class TestTwoSumsHashed(unittest.TestCase):
    """Набор тестов для функции two_sums_hashed."""

    def test_simple(self):
        """Обычный пример: одна подходящая пара."""
        self.assertEqual(two_sums_hashed([1, 2, 3, 4, 5, 6, 7, 8, 9], 8), [(2,4), (1,5), (0, 6)])

    def test_multiple_pairs(self):
        """Несколько пар — возвращаются все."""
        self.assertEqual(
            two_sums_hashed([2, 1, 4, 5, 2], 6),
            [(0, 2), (1, 3), (2, 4)],
        )

    def test_repeated_values(self):
        """Повторяющиеся значения: все комбинации индексов."""
        self.assertEqual(
            two_sums_hashed([1, 4, 4, 4, 4], 8),
            [(1, 2), (1, 3), (2, 3), (1, 4), (2, 4), (3, 4)],
        )

    def test_triple_same_number(self):
        """Тройное повторение: пары со всеми комбинациями индексов."""
        self.assertEqual(
            two_sums_hashed([3, 3, 3], 6),
            [(0, 1), (0, 2), (1, 2)],
        )

    def test_no_pair(self):
        """Пары нет — возвращается пустой список."""
        self.assertEqual(two_sums_hashed([1, 2, 3, 4], 100), [])

    def test_empty_list(self):
        """Пустой список — возвращается пустой список."""
        self.assertEqual(two_sums_hashed([], 5), [])

    def test_single_element(self):
        """Список из одного элемента — пустой список."""
        self.assertEqual(two_sums_hashed([5], 10), [])

    def test_negative_numbers(self):
        """Отрицательные числа, target = 0."""
        # -3 + 3 = 0 (0,3); -1 + ? нет; 2 + (-2) нет
        self.assertEqual(two_sums_hashed([-3, -1, 2, 3, 4], 0), [(0, 3)])

    def test_target_zero_with_zeros(self):
        """Нули дают пары сами с собой."""
        # Пары: 0+0=0 → все индексы нулей между собой
        self.assertEqual(
            two_sums_hashed([0, 0, 0], 0),
            [(0, 1), (0, 2), (1, 2)],
        )


if __name__ == '__main__':
    unittest.main() # для VS Code
```
