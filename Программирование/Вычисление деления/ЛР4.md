```python
import math
from configparser import ConfigParser
def calculate(op1:int|float, op2: int|float, epsilon: float = 0.0001):
    if op2 == 0:
        raise ZeroDivisionError("Деление на ноль невозможно")
    n = round(-math.log(epsilon, 10))
    return round(op1/op2,n)

def load_params(path="settings.ini"):
    parser = ConfigParser()
    parser.read("settings.ini")
    if not parser.has_section("calculation"):
            return None

    try:                                        # 5. прочитать значение
        value = parser.getfloat("calculation", "epsilon")
    except ValueError:
        return None
    if 10**-9<=value<=10**-1:
        return value
    else:
        return None


"""Вне функции создадим конфигурационный файл"""
with open ("settings.ini", "w") as f:
    f.write("[calculation]\n")
    f.write("epsilon = 0.000001\n")
print (calculate(1, 3, load_params("settings.ini")))
```
