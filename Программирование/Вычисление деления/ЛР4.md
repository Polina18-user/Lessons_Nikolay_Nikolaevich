```python
import math
from configparser import ConfigParser
import logging
"""Модуль использует стандартный логгер 'myapp' для записи сообщения об ошибках
Глобальная настройка логирования: уровень ERROR формат сообщения включает время, имя логера, уровень и сообщение
"""
logger = logging.getLogger('myapp')
logging.basicConfig(
    level=logging.ERROR,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
)


def calculate(op1: int | float, op2: int | float, epsilon: float = 0.0001):
    """
    Делит op1 на op2 и округляет результат с точностью epsilon.

    :param op1: Делимое
    :type op1: int | float

    :param op2: Делитель
    :type op2: int | float

    :param epsilon: 
    :type epsilon: float
    :return: Результат  деления op1/op2Б округленный до n знаков
    :rtype: float
    :raises ZeroDivisionError: Если op2 равен нулю
    """
    if op2 == 0:
        error_text = "Деление на ноль"
        logger.error(error_text)
        raise ZeroDivisionError("Деление на ноль невозможно")

    n = round(-math.log(epsilon, 10))
    return round(op1 / op2, n)


def load_params(path: str="settings.ini", section: str="calculation", option: str="epsilon") -> float:
    '''читает параметр epsilon из  INI-файла
    :param path: Путь к ini-файлу. По умолчанию 'settings.ini'
    :type path: str
    
    :param section: Имя секции в файле. По умолчанию 'calculation'
    :type section: str

    :param option: Имя параметра в секции. по умолчанию 'epsilon'
    :type option: str

    :return Значение параметра epsilon из файла.
    :rtype: float
    :raises FileNotFoundError: Если файл не найден или недоступен для чтения
    :raises KeyError: Если в файле отсутсвует указанная секция или параметр
    :raises ValueError: Если значения параметра не являются числом или выходит за допустимый диапазон.
    
    '''
    parser = ConfigParser()
    read_files = parser.read(path, encoding="utf-8")

    if not read_files:
        file_not_found_error_text = f"Файл {path!r} не найден или недоступен для чтения"
        logger.critical(file_not_found_error_text)
        raise FileNotFoundError(file_not_found_error_text)

    if not parser.has_section(section):
        raise KeyError(f"В файле нет секции [{section}]")

    if not parser.has_option(section, option):
        raise KeyError(f"В секции [{section}] нет параметра {option!r}")

    try:
        epsilon = parser.getfloat(section, option)
    except ValueError:
        raise ValueError(f"Значение {option} не является числом!")

    if not (10 ** -9 <= epsilon <= 10 ** -1):
        raise ValueError(f"Значение {option} не попадает в диапазон!")

    return epsilon

# Демонстрация работы логгера на разных уровнях
if __name__ == "__main__":
    logging.debug("Это сообщение для отладки")
    logging.info("Информационное сообщение")
    logging.warning("Предупреждение!")
    logging.error("Произошла ошибка")
    logging.critical("Критическая ошибка")

    #Создаем файл настроек с валидным значение epsilon
    with open("settings.ini", "w") as f:
        f.write("[calculation]\n")
        f.write("epsilon = 0.000001\n")

    print(calculate(1, 3, load_params("settings.ini")))
```

'''python
import os
import tempfile
import unittest

from devision import calculate, load_params

class TestCalculate(unittest.TestCase):
    def test_half_with_epsilon_01(self):
        result = calculate(1, 2, epsilon=0.1)
        self.assertAlmostEqual(result, 0.5, places=1)

    def test_one_over_thousand(self):
        result = calculate(1,1000, epsilon = 0.001)
        self.assertAlmostEqual(result, 0.001, places=3)


    def test_division_by_zero(self):
        self.assertRaises(ZeroDivisionError, calculate, 1, 0)

class TestLoadParams(unittest.TestCase):

    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.path = os.path.join(self.tmp.name, "settings.ini")

    def tearDown(self):
        self.tmp.cleanup()

    def _write_settings(self, text):
        with open(self.path, "w", encoding="utf-8") as file:
            file.write(text)

    def test_load_valid_epsilon(self):
        self._write_settings("[calculation]\nepsilon = 0.001\n")

        epsilon = load_params(self.path)

        self.assertEqual(epsilon, 0.001)

    def test_file_not_found(self):
        missing_path = os.path.join(self.tmp.name, "missing.ini")

        with self.assertRaises(FileNotFoundError):
            load_params(missing_path)

    def test_epsilon_lower_bound(self):
        self._write_settings("[calculation]\nepsilon = 0.000000001\n")  # 10**-9
        self.assertEqual(load_params(self.path), 10 ** -9)

    def test_epsilon_upper_bound(self):
        self._write_settings("[calculation]\nepsilon = 0.1\n")  # 10**-1
        self.assertEqual(load_params(self.path), 0.1)

    def test_epsilon_out_of_range(self):
        self._write_settings("[calculation]\nepsilon = 0.5\n")

        with self.assertRaises(ValueError):
            load_params(self.path)

    def test_invalid_number_format(self):
        self._write_settings("[calculation]\nepsilon = abc\n")

        with self.assertRaises(ValueError):
            load_params(self.path)

    def test_missing_section(self):
        self._write_settings("[other]\nepsilon = 0.001\n")

        with self.assertRaises(KeyError):
            load_params(self.path)

    def test_missing_option(self):
        self._write_settings("[calculation]\nother = 0.001\n")

        with self.assertRaises(KeyError):
            load_params(self.path)

if __name__ == "__main__":
    unittest.main()
'''
