# Лабораторная работа №2. Построение бинарного дерева (рекурсией).

## Математическая модель
14) Root = 14; height = 4, left_leaf = 2-(root-1), right_leaf = root*2

## Код программы
```python
from typing import Callable

def left_branch(root: int) -> int:
    """
    Находит значение левого от корня элемента по формуле
    :param root: само значение корня
    :type root: int
    :rtype: int
    """
    return 2-(root-1)

def right_branch(root: int) -> int:
    """
    Находит значение правого от корня элемента по формуле
    :param root: само значение корня
    :type root: int
    :rtype: int
    """
    return root*2

def gen_bin_tree(height: int = 4, root: int = 14, l_b: Callable[[int], int]=left_branch, r_b: Callable[[int], int]=right_branch) ->dict:
    """
    Рекурсивно строит бинарное дерево в виде вложенных словарей

    Каждый узед имеет вид {root : {"left" : ..., "right": ...}}, лист - {root:None}.
    Значение потомков вычисляются функциями l_b и r_b.

    :param height: высота дерева (0 - только корень)
    :type height: int
    
    :param root: значение корня
    :type root: int

    :param l_b: функция вычисления левого потомка
    :type l_b: Callable[[int], int]

    :param r_b: функция вычисления правого потомка
    :type r_b: Callable[[int], int]

    :return: дерево в виде вложенных словарей
    :rtype: dict   
    """
    if height == 0: #проверяем случай, когда root - лсит
        return {root:None}
    left_side = l_b(root) #Присваиваем левого потомка
    right_side = r_b(root) #Присваиваем правого потомка
    
    left_sub = gen_bin_tree(height-1, left_side, l_b, r_b) #рекурсивно вызываем для создания левого дерева
    right_sub = gen_bin_tree(height-1, right_side, l_b, r_b)  #рекурсивно вызываем для создания правого дерева
    res={root:{'left':left_sub, 'right': right_sub}} #красиво записываем
    
    return res


n = int(input("Сколько вершин? "))
roots = []
for _ in range(n):
    roots.append(int(input("Введите корень: ")))

for root in roots:
    print(f"Дерево для корня {root}:")
    print(gen_bin_tree(height=2, root=root))
    print()   # пустая строка для читаемости
```
