### binary_tree_without_recursia.py
```python
from typing import Callable
from collections import deque

def left_branch(root: int) -> int:
    """
        Находит значение левого от корня элемента по формуле
        :param root: само значение корня
        :type root: int
        :rtype: int

    """
    return 2-(root-1)

def right_branch(root: int) -> int:
    """
    Находит значение правого от корня элемента по формуле
    :param root: само значение корня
    :type root: int
    :rtype: int
    """
    return root*2


def gen_bin_tree_without(height: int = 4, root: int = 14,l_b: Callable [[int], int] = left_branch, r_b: Callable [[int], int] = right_branch ) -> dict:
    """
    Функция нерукрсивного создания дерева
    :param height: высота дерева
    :type height: int

    :param root: высота дерева
    :type root: int

    :param height: высота дерева
    :type height: int

    :param l_b: функция вычисления левого потомка
    :type l_b: Callable[[int], int]
    
    :param r_b: функция вычисления правого потомка
    :type r_b: Callable[[int], int]
    
    :return: дерево в виде вложенных словарей
    :rtype: dict   

    """

    if height == 0:
        return {root:None} #если это лист - то возвращаем словарь без левого и правого
    else:
        tree = {root:{"left": None, "right": None}} #если не лист, то делаем заготовку дерева
        queue = deque([(root, height, tree)]) # создание создание двухсторонней очереди
        while queue: #пока очередь существует
            value, h, node = queue.popleft() # снимаем слева

            left_val = l_b(value)
            right_val = r_b(value)

            if h-1==0: #если это не предпоследний уровень
                node[value]["left"] = {left_val: None} #делаем заготовки дерева
                node[value]["right"]={right_val: None}
            else:
                node[value]["left"] = {left_val: {"left": None, "right": None}}
                node[value]["right"] = {right_val: {"left": None, "right": None}}
                queue.append((left_val, h-1, node[value]["left"]))#добавляем в очередь новый корень, новую высоту и новое дерево
                queue.append((right_val, h-1, node[value]["right"]))
        return tree
print(gen_bin_tree_without())
```

"""
Тесты для функции gen_bin_tree из модуля binary_tree.
"""
import unittest
from binary_tree_without_recursia import gen_bin_tree_without


class TestGenBinTree(unittest.TestCase):
    """Набор тестов для функции gen_binary_tree_without."""

    def test_0(self):
        """Высота 0 - только корень"""
        self.assertEqual(gen_bin_tree_without(height = 0, root = 14),{14: None})

    def test_1(self):
        """Высота 1"""
        self.assertEqual(gen_bin_tree_without(height = 1, root = 14),{14:{"left":{-11:None},"right": {28:None}}})
    def test_2(self):
        """Высота 2"""
        self.assertEqual(gen_bin_tree_without(height = 2, root = 14),{14: {'left': {-11: {'left': {14: None}, 'right': {-22: None}}}, 'right': {28: {'left': {-25: None}, 'right': {56: None}}}}})


if __name__ == '__main__':
    unittest.main() # для VS Code
